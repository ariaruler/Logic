\section{What is proposition?}

\begin{definition}
    \textbf{predicate}  ``$set()$'' and identity ``$=$'' and membership ``$\in$'' symbols are predicates.
    ``$set()$'' is unary and identity and membership  are binary.
\end{definition}

capital letter ``$(P,Q,R)$'' showes \textbf{predicates}.

%
\begin{definition}
    \textbf{Atomic proposition} we define atomic propositions like this.

    \begin{enumerate}
        \setlength{\itemsep}{0pt} % No extra space between items
        \item if ``$a$'' is an obj then ``$set(a)$'' is an atomic proposition.
        \item if ``$a,b$'' are obj then ``$a=b$'' is an atomic proposition.
        \item if ``$a,b$'' are obj then ``$a \in b$'' is an atomic proposition.
    \end{enumerate}
    we denote atomic propositions with ``$p_{i}$''.
\end{definition}

%
\begin{undefined}
    \textbf{logical operators} ($\neg ,\to ,\land ,\lor$) tool that used between
    atoms to make more complex propositions
    (all of the opratores are binary except "$\neg$" that is unary ).
\end{undefined}

you can easily see that formula are made up by atomic propositions.in general
we say formula is an atomic formula or combination of them.

%
\begin{definition}
    \textbf{Propositions} these are propositions.

    \begin{enumerate}
        \setlength{\itemsep}{0pt} % No extra space between items
        \item atomic propositions ``$(p_{i})$''.
        \item if ``$p$'' is a formula then ``$\neg p$'' is a formula too.
        \item if ``$p ,q$'' are two propositions then ``$p \bigcirc q $'' is a formula
              too.($\bigcirc $ can be ``$\to,\land ,\lor$'').
    \end{enumerate}

\end{definition}
for example ``$(p_0 \to p_1 ,\neg p_3)$'' \\
we denote propositions with ``$p ,q ,r ,...$''.

\begin{definition}

\begin{definition}

    \end{enumerate}

\begin{definition}
    \textbf{Substitution for formulas} we define Substitution for a formula like this.
    \begin{enumerate}
        \setlength{\itemsep}{0pt} % No extra space between items
        \item for atomic formulas we define it like this. if it's a single object we do it as
              we explain in ``\textbf{Substitution for obj}''. if it's like $x = y$ or $x \in
                  y$ ($x,y$ can be both constants and variables). we define it like this.
              \[ Set(y)[a/x] := Set(y[a/x]) \]
              \[ (x=y)[a/z] := (x[a/z] = y[a/z]) \]
              \[ (x \in y)[a/z] := (x[a/z] \in y[a/z]) \]

              \[ (B \bigcirc C)[a/x] := (B[a/x] \bigcirc C[a/x]) \]

              \[ (\neg B)[a/x] := (\neg B[a/x]) \]
    \end{enumerate}
\end{definition}

\begin{definition}
    \textbf{Function from set to set}  we call a function \textbf{f} from \textbf{A} to \textbf{B}
    when it's domain is  \textbf{A} and it's range is subset of \textbf{B} i.e.
    \[ (f:A \rightarrow B) : Func(f) \land Dom(f)=A \land Range(f) \subseteq B \]
\end{definition}

\begin{definition}
    \textbf{Set of functions from set to set}  we denote set of all ``$f:A \rightarrow B$'' with ``$B^A$''  i.e.
    \[ B^A : \{ f \mid f :A \rightarrow B \} \]
\end{definition}

\begin{definition}
    \textbf{Injective function}
    we call a function \textbf{f} injective when every output has at most one input i.e.
    \[ Injective(f) : \forall y ,x_1 , x_2 \Big((x_1,y) \in f \wedge (x_2,y) \in f \rightarrow x_1 = x_2 \Big). \]

    we call a function \textbf{f} injective from \textbf{A} to \textbf{B} when it's
    function from \textbf{A} to \textbf{B} and it's injective i.e.
    \[ (f:A \xrightarrow{inj} B) :  (f:A \rightarrow B) \land Injective(f) \]
\end{definition}

\begin{definition}
    \textbf{Surjective function}  we call a function \textbf{f} from \textbf{A} to \textbf{B}
    when it's domain is  \textbf{A} and it's range is equal to \textbf{B} i.e.
    \[ (f:A \xrightarrow[surj]{} B) :  (f:A \rightarrow B)  \land Range(f) = B \]
    or i.e.
    \[ (f:A \xrightarrow[surj]{} B) :  (f:A \rightarrow B)  \land (\forall y \in B \quad \exists x \in A \quad (x,y) \in f) \]
\end{definition}

\begin{definition}
    \textbf{Bijection function}  we call a function \textbf{f} from \textbf{A} to \textbf{B}
    when it's domain is  \textbf{A} and it's range is equal to \textbf{B} i.e.
    \[ (f:A \xrightarrow[surj]{inj} B) :  (f:A \xrightarrow{inj} B)  \land (f:A \xrightarrow[surj]{} B) \]
\end{definition}

\section{Structures and morphisms}

\begin{definition}
    \textbf{Structure} ``$\mathfrak{M} $'' is a tuple defined like this.
    \[\mathfrak{M}  := (M , f_{0}, f_{1},..., f_{n},R_{0}, R_{1} ,...,R_{m}, c_{0}, c_{1},..., c_{k}) \]
    ``$M$'' is a \textbf{nonempty} set shows our universe ``$f_i$'' are funcions and ``$R_I$'' are relations and ``$c_i$'' are constants.
    and for the case we have arbitary family of functions and relations and constants.
    \[\mathfrak{M}  := (M , \{f_i \}_{i \in I_f},\{R_i \}_{i \in I_R}, \{c_i \}_{i \in I}) \]
\end{definition}

\begin{definition}
    \textbf{Homorphism} if  ``$\mathfrak{M} := (M , \{f_{i,m} \}_{i \in I_f},\{R_{i,m} \}_{i \in I_R}, \{c_{i,m} \}_{i \in I}) $''
    and ``$\mathfrak{N} := (N , \{f_{i,n} \}_{i \in I_f},\{R_{i,n} \}_{i \in I_R}, \{c_{i,n} \}_{i \in I}) $'' are two structures.
    function ``$h : M \to N$''is called ``(weak) Homorphism'' when this conditions holds
    \begin{enumerate}
        \setlength{\itemsep}{0pt} % No extra space between items
        \item for eveery relation ``$R_{i,m}$''
              \[ \forall x_0 , x_1 , x_2 , ... , x_k \in M \quad R_{i,m}(x_0 , x_1 , x_2 , ... , x_k) \to R_{i,n}(h(x_0),h(x_1) ,h(x_2), ... ,h(x_k)) \].
        \item for eveery function ``$f_{i,m}$''
              \[ \forall x_0 , x_1 , x_2 , ... , x_k \in M \quad h(f_{i,m}(x_0 , x_1 , x_2 , ... , x_k)) = f_{i,n}(h(x_0),h(x_1) ,h(x_2), ... ,h(x_k)) \].
        \item for eveery constant ``$c_{i,m}$''
              \[  h(c_{i,m}) = c_{i,n} \].
    \end{enumerate}
    in condition (1) if ``$\leftrightarrow$'' be instead of ``$\to $'' then it's called  ``(strong) Homorphism''
\end{definition}

few notices: when we are talking about ``Homorphism'' we usually talk about
``(strong) Homorphism'' .\\ in the case (in definition ``(strong) Homorphism'')
we denote it by ``$Hom(h,\mathfrak{M} , \mathfrak{N} )$''. if exists function
``$h$'' ``$Hom(h,\mathfrak{M} , \mathfrak{N} )$'' we say ``$\mathfrak{M}$'' and
``$\mathfrak{N}$'' are homomorphic. we denote it by ``$Hom(\mathfrak{M} ,
    \mathfrak{N} )$''.\\ if ``$h : M \xrightarrow[]{inj} N$'' we call it
``Monomorphism'' or ``Embedding''.we say ``$\mathfrak{M}$'' and
``$\mathfrak{N}$'' are momomorphic.\\ if ``$h : M \xrightarrow[surj]{} N$'' we
call it ``Epimorphism''.we say ``$\mathfrak{M}$'' and ``$\mathfrak{N}$'' are
Eomomorphic.\\ if ``$h : M \xrightarrow[surj]{inj} N$'' we call it
``Isomorphism''.we say ``$\mathfrak{M}$'' and ``$\mathfrak{N}$'' are
Iomomorphic. and denote it by ``$ \mathfrak{M} \cong \mathfrak{N}$''.\\ if ``$k
    : M \to M$'' is a function and ``$Hom(k,\mathfrak{M},\mathfrak{M})$''. we call
``$k$'' ``Indomorphism''.\\ if ``$k : M \xrightarrow[surj]{inj} M$'' is a
bijection and ``$Hom(k,\mathfrak{M},\mathfrak{M})$''. we call ``$k$''
``Automorphism''.\\

\begin{definition}

    \[ (asymmetric) :   \forall x,y \in A \quad x \prec y \to y \nprec x \]
