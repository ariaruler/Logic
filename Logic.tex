\section{What is proposition?}

\begin{definition}
    \textbf{predicate}  ``$set()$'' and identity ``$=$'' and membership ``$\in$'' symbols are predicates.
    ``$set()$'' is unary and identity and membership  are binary.
\end{definition}

capital letter ``$(P,Q,R)$'' showes \textbf{predicates}.

%
\begin{definition}
    \textbf{Atomic proposition} we define atomic propositions like this.

    \begin{enumerate}
        \setlength{\itemsep}{0pt} % No extra space between items
        \item if ``$a$'' is an obj then ``$set(a)$'' is an atomic proposition.
        \item if ``$a,b$'' are obj then ``$a=b$'' is an atomic proposition.
        \item if ``$a,b$'' are obj then ``$a \in b$'' is an atomic proposition.
    \end{enumerate}
    we denote atomic propositions with ``$p_{i}$''.
\end{definition}

%
\begin{undefined}
    \textbf{logical operators} ($\neg ,\to ,\land ,\lor$) tool that used between
    atoms to make more complex propositions
    (all of the opratores are binary except "$\neg$" that is unary ).
\end{undefined}

you can easily see that formula are made up by atomic propositions.in general
we say formula is an atomic formula or combination of them.

%
\begin{definition}
    \textbf{Propositions} these are propositions.

    \begin{enumerate}
        \setlength{\itemsep}{0pt} % No extra space between items
        \item atomic propositions ``$(p_{i})$''.
        \item if ``$p$'' is a formula then ``$\neg p$'' is a formula too.
        \item if ``$p ,q$'' are two propositions then ``$p \bigcirc q $'' is a formula
              too.($\bigcirc $ can be ``$\to,\land ,\lor$'').
    \end{enumerate}

\end{definition}
for example ``$(p_0 \to p_1 ,\neg p_3)$'' \\
we denote propositions with ``$p ,q ,r ,...$''.

\begin{definition}

\begin{definition}
    \textbf{Substitution for formulas} we define Substitution for a formula like this.
    \begin{enumerate}
        \setlength{\itemsep}{0pt} % No extra space between items
        \item for atomic formulas we define it like this. if it's a single object we do it as
              we explain in ``\textbf{Substitution for obj}''. if it's like $x = y$ or $x \in
                  y$ ($x,y$ can be both constants and variables). we define it like this.

              \[ (x \in y)[a/z] := (x[a/z] \in y[a/z]) \]

        \item   $ \forall x(P(x) \wedge Q(x)) \therefore / \therefore \forall x P(x) \wedge \forall x Q(x) $ (D F $\forall \land$)(Distribution and Factorization of $\forall \land$)
        \item   $ \exists x(P(x) \vee Q(x)) \therefore /\therefore \exists x P(x) \vee \exists x Q(x) $ (D F $\exists \lor$)
        \item   $ \forall x P(x) \vee \forall x Q(x) /\therefore \forall x(P(x) \vee Q(x)) $   (F $\forall \lor$)
        \item   $ \exists x(P(x) \wedge Q(x)) /\therefore \exists x P(x) \wedge \exists x Q(x) $   (D $\exists \land$)
        \item   $ \forall x(P(x) \rightarrow Q(x)) /\therefore  \forall x P(x) \rightarrow \forall x Q(x) $  (D $\forall \to$)
        \item   $\exists x P(x) \rightarrow \exists x Q(x) /\therefore \exists x(P(x) \rightarrow Q(x)) $  (F $\exists \to$)
        \item   $ \forall x P(x) \wedge Q \therefore /\therefore \forall x(P(x) \wedge Q) $

        \item   $ \exists x P(x) \wedge Q \therefore /\therefore \exists x(P(x) \wedge Q) $
        \item   $ \exists x P(x) \vee Q \therefore /\therefore \exists x(P(x) \vee Q) $
        \item   $Q \rightarrow \forall x P(x) \therefore /\therefore \forall x(Q \rightarrow P(x)) $
        \item   $Q \rightarrow \exists x P(x) \therefore /\therefore \exists x(Q \rightarrow P(x)) $
        \item   $\forall x P(x) \rightarrow Q \therefore /\therefore \exists x(P(x) \rightarrow Q) $

\begin{definition}
    \textbf{Relation} we call a set like \textbf{R} a \textbf{Relation} when there are
    two sets like $X$ and $Y$ s.t. $R \subseteq X \times Y$.
    \[ Rel(R) : \exists X,Y (R \subseteq X \times Y). \]
\end{definition}

sometimes instead of ``$(x,y) \in R$'' we write ``$xRy $''.

\begin{definition}
    \textbf{Model} (this is a temporary definition, we will give a better one in the next chapter.)
    ``$\mathfrak{M} $'' is a tuple defined like this.
    \[\mathfrak{M}  := (M ,R_{0}, R_{1} ,...,R_{n}, \{c_i \}_{i \in I}) \]
    ``$M$'' is a set shows our universe  and ``$R_I$'' are relations and ``$c_i$'' are constants.
    and for the case we have infinite relations.
    \[\mathfrak{M}  := (M ,\{R_i \}_{i \in I}, \{c_i \}_{i \in I}) \]

\end{definition}

\begin{definition}
    \textbf{ِDomain and Range(also called Image)} \\
    Domain of a set ``$R$'' is all ``$x$'' that there exists a ``$y$'' s.t. ``$xRy$''
    \[ Dom (R) := \{ x \mid \exists y  \quad xRy\} \]
    Range(Image) of a set ``$R$'' is all ``$y$'' that there exists a ``$x$'' s.t.
    ``$xRy$''
    \[ Range (R) := \{ y  \mid \exists x \quad xRy\}. \]
\end{definition}

\begin{definition}

\begin{definition}

\end{definition}

\begin{definition}

    ``output of a function'' is object too, so we can have $f(f(x))$  as well.
\end{definition}

\begin{definition}
    \textbf{Function from set to set}  we call a function \textbf{f} from \textbf{A} to \textbf{B}
    when it's domain is  \textbf{A} and it's range is subset of \textbf{B} i.e.
    \[ (f:A \rightarrow B) : Func(f) \land Dom(f)=A \land Range(f) \subseteq B \]
\end{definition}

\begin{definition}
    \textbf{Injective function}
    we call a function \textbf{f} injective when every output has at most one input i.e.
    \[ Injective(f) : \forall y ,x_1 , x_2 \Big((x_1,y) \in f \wedge (x_2,y) \in f \rightarrow x_1 = x_2 \Big). \]

    we call a function \textbf{f} injective from \textbf{A} to \textbf{B} when it's
    function from \textbf{A} to \textbf{B} and it's injective i.e.
    \[ (f:A \xrightarrow{inj} B) :  (f:A \rightarrow B) \land Injective(f) \]
\end{definition}

\begin{definition}
    \textbf{Surjective function}  we call a function \textbf{f} from \textbf{A} to \textbf{B}
    when it's domain is  \textbf{A} and it's range is equal to \textbf{B} i.e.
    \[ (f:A \xrightarrow[surj]{} B) :  (f:A \rightarrow B)  \land Range(f) = B \]
    or i.e.
    \[ (f:A \xrightarrow[surj]{} B) :  (f:A \rightarrow B)  \land (\forall y \in B \quad \exists x \in A \quad (x,y) \in f) \]
\end{definition}

\begin{definition}
    \textbf{Bijection function}  we call a function \textbf{f} from \textbf{A} to \textbf{B}
    when it's domain is  \textbf{A} and it's range is equal to \textbf{B} i.e.
    \[ (f:A \xrightarrow[surj]{inj} B) :  (f:A \xrightarrow{inj} B)  \land (f:A \xrightarrow[surj]{} B) \]
\end{definition}

\begin{definition}

\begin{definition}

    \[ \bigcup_{i \in I} A_i := \bigcup \{ A_i \}_{i \in I}  \]

\begin{definition}
    \textbf{Natural numbers} we define ``$0,1,2,3,...$'' like this.
    \begin{align*}
         &
        0 := \emptyset
        \\
         &
        1 :=\{0\}= \emptyset \cup \{ \emptyset \} = \{ \emptyset \}
        \\
         &
        2 := \{0,1\}=\{ \emptyset \} \cup \big\{ \{ \emptyset \} \big\} = \big\{ \emptyset , \{ \emptyset \} \big\}
        \\
         &
        3 := \{0,1,2\}=\Big\{ \emptyset , \{ \emptyset \},\big\{ \emptyset , \{ \emptyset \} \big\} \Big\} \\
         & \vdots
    \end{align*}

\begin{definition}
    \textbf{Successor function} is a function ``$s :V \to V$'' defined like this.
    \[s(x) := x\cup \{x\}\]
\end{definition}
